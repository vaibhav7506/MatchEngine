cmake_minimum_required(VERSION 3.20)
project(MatchEngine VERSION 1.0.0 LANGUAGES CXX)

option(MATCHENGINE_BUILD_TESTS "Build Catch2 unit tests" ON)
option(MATCHENGINE_BUILD_BENCHMARK "Build the standalone benchmark" ON)
option(MATCHENGINE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(MATCHENGINE_BUILD_PYTHON "Build the optional pybind11 extension" OFF)
option(MATCHENGINE_BUILD_ITCH "Build the standalone ITCH parser and CLI" ON)

add_library(matchengine src/engine.cpp)
add_library(MatchEngine::matchengine ALIAS matchengine)
target_include_directories(matchengine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(matchengine PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(matchengine PRIVATE /W4 /permissive-)
else()
  target_compile_options(matchengine PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

if(MATCHENGINE_BUILD_ITCH)
  add_library(itch_parser src/itch/parser.cpp src/itch/reconciler.cpp)
  add_library(MatchEngine::itch_parser ALIAS itch_parser)
  target_include_directories(itch_parser PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(itch_parser PUBLIC cxx_std_17)
  if(MSVC)
    target_compile_options(itch_parser PRIVATE /W4 /permissive-)
  else()
    target_compile_options(itch_parser PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    if(MATCHENGINE_SANITIZERS)
      target_compile_options(itch_parser PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(itch_parser PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
  add_executable(itch_stats tools/itch_stats.cpp)
  target_link_libraries(itch_stats PRIVATE MatchEngine::itch_parser)
  if(MATCHENGINE_BUILD_TESTS)
    enable_testing()
    add_executable(itch_tests tests/test_itch.cpp)
    target_include_directories(itch_tests SYSTEM PRIVATE third_party)
    target_link_libraries(itch_tests PRIVATE MatchEngine::itch_parser)
    add_test(NAME itch_tests COMMAND itch_tests)
  endif()
endif()

if(MATCHENGINE_BUILD_PYTHON)
  # Python builds explicitly opt in. Ordinary Phase 1 builds remain offline.
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
    OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE
    COMMAND_ERROR_IS_FATAL ANY)
  find_package(pybind11 3.0.1 CONFIG REQUIRED)
  set_target_properties(matchengine PROPERTIES POSITION_INDEPENDENT_CODE ON)
  pybind11_add_module(matchengine_py bindings/module.cpp)
  target_link_libraries(matchengine_py PRIVATE MatchEngine::matchengine)
  install(TARGETS matchengine_py LIBRARY DESTINATION matchengine RUNTIME DESTINATION matchengine)
  if(MINGW)
    # CPython 3.8+ resolves DLLs from the extension directory, not just PATH.
    get_filename_component(MATCHENGINE_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(runtime libc++.dll libunwind.dll libwinpthread-1.dll)
      if(EXISTS "${MATCHENGINE_COMPILER_BIN}/${runtime}")
        install(FILES "${MATCHENGINE_COMPILER_BIN}/${runtime}" DESTINATION matchengine)
      endif()
    endforeach()
  endif()
endif()
if(MATCHENGINE_SANITIZERS AND NOT MSVC)
  target_compile_options(matchengine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(matchengine PUBLIC -fsanitize=address,undefined)
endif()

if(MATCHENGINE_BUILD_TESTS)
  enable_testing()
  add_executable(matchengine_tests tests/test_engine.cpp)
  target_include_directories(matchengine_tests SYSTEM PRIVATE third_party)
  target_link_libraries(matchengine_tests PRIVATE MatchEngine::matchengine)
  add_test(NAME matchengine_tests COMMAND matchengine_tests)
endif()

if(MATCHENGINE_BUILD_BENCHMARK)
  add_executable(matchengine_benchmark benchmarks/benchmark.cpp)
  target_link_libraries(matchengine_benchmark PRIVATE MatchEngine::matchengine)
  if(WIN32)
    target_link_libraries(matchengine_benchmark PRIVATE psapi)
  endif()
endif()
